file(GLOB_RECURSE TESTS CONFIGURE_DEPENDS *.f90 *.F90)

# Test programs that need TecIO are built only when ORION is built with TecIO.
if(NOT USE_TECIO)
  list(FILTER TESTS EXCLUDE REGEX "/(tecplot_read_szplt_plane_xyz|tecplot_read_szplt_long_header)[.]f90$")
endif()

set(TESTS ${TESTS} PARENT_SCOPE)

set(EXECUTABLE_OUTPUT_PATH "${CMAKE_SOURCE_DIR}/bin/test")

# Runtime directory shared by the integration tests.
set(TEST_RUNTIME_DIR "${CMAKE_BINARY_DIR}/test-runtime")
file(MAKE_DIRECTORY "${TEST_RUNTIME_DIR}")

foreach(TEST ${TESTS})

  get_filename_component(EXE_NAME ${TEST} NAME_WE)

  add_executable(${EXE_NAME} ${TEST})

  set_target_properties(${EXE_NAME} PROPERTIES
    LINKER_LANGUAGE Fortran
  )

  target_link_libraries(${EXE_NAME} ORION)

  if(USE_TECIO)
    target_link_libraries(${EXE_NAME} ${LINKLIBS})
  endif()

  target_include_directories(${EXE_NAME}
    INTERFACE
      $<BUILD_INTERFACE:${CMAKE_Fortran_MODULE_DIRECTORY}>
  )

  if(BUILD_TESTING)
    add_test(
      NAME ORION.${EXE_NAME}
      COMMAND ${EXE_NAME}
    )

    set_tests_properties(
      ORION.${EXE_NAME}
      PROPERTIES
        WORKING_DIRECTORY "${TEST_RUNTIME_DIR}"
    )
  endif()

endforeach()

# tecplot_write and tecplot_read_strandid check the files of the binary writer, tecplot_read reads binary files and
# converter_app converts to and from one only when ORION is built with TecIO; without it converter_app checks that the
# converter, asked for a binary file, ends with a nonzero exit status.
if(USE_TECIO)
  target_compile_definitions(tecplot_write PRIVATE TECIO)
  target_compile_definitions(tecplot_read PRIVATE TECIO)
  target_compile_definitions(tecplot_read_strandid PRIVATE TECIO)
  target_compile_definitions(converter_app PRIVATE TECIO)
endif()

# converter_app runs the converter (src/fortran/app), whose path it takes from ORION_CONVERTER.
set_tests_properties(
  ORION.converter_app
  PROPERTIES
    ENVIRONMENT "ORION_CONVERTER=$<TARGET_FILE:converter>"
)

# Some read tests consume files generated by the corresponding write tests.
set_tests_properties(
  ORION.p3d_read
  PROPERTIES
    DEPENDS ORION.p3d_write
)

set_tests_properties(
  ORION.tecplot_read
  PROPERTIES
    DEPENDS ORION.tecplot_write
)

# vtk_read_wrapper has a time limit: one of its reads took about 15 minutes when the reader looked for field data through a
# block file that holds none. The program takes less than a second on a local disk; the limit leaves room for a slow one.
set_tests_properties(
  ORION.vtk_read_wrapper
  PROPERTIES
    DEPENDS ORION.vtk_write_wrapper
    TIMEOUT 300
)

# The Python tests need a Python 3 with NumPy; without one they are not registered.
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  execute_process(
    COMMAND ${Python3_EXECUTABLE} -c "import numpy"
    RESULT_VARIABLE NUMPY_IMPORT_RESULT
    OUTPUT_QUIET ERROR_QUIET
  )
endif()
if(Python3_Interpreter_FOUND AND NUMPY_IMPORT_RESULT EQUAL 0)
  # read_P3D reads the PLOT3D layouts of p3d_read_formats, written by the test itself.
  add_test(
    NAME ORION.python_read_p3d
    COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_SOURCE_DIR}/src/python/test/read_p3d.py
  )
  set_tests_properties(
    ORION.python_read_p3d
    PROPERTIES
      WORKING_DIRECTORY "${TEST_RUNTIME_DIR}"
  )
  # The Python reader reads back the Tecplot ASCII file of tecplot_write, which
  # writes it only with TecIO (its binary writes come first).
  if(USE_TECIO)
    add_test(
      NAME ORION.python_read_tec
      COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_SOURCE_DIR}/src/python/test/read_tec.py
    )
    set_tests_properties(
      ORION.python_read_tec
      PROPERTIES
        WORKING_DIRECTORY "${TEST_RUNTIME_DIR}"
        DEPENDS ORION.tecplot_write
    )
  endif()
else()
  message(STATUS "No Python 3 with NumPy: the Python tests (ORION.python_*) are not registered")
endif()
